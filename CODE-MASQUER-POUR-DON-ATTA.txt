<script>
// Masquer la section "Mode d'expédition" UNIQUEMENT si le produit Don ATTA est dans le panier
(function() {
    'use strict';
    
    function hideShippingForDonation() {
        // Vérifier qu'on est sur checkout
        if (!window.location.href.includes('/checkout')) return;
        
        // Vérifier si le panier contient un produit Don ATTA
        fetch('/cart.js')
            .then(function(response) {
                return response.json();
            })
            .then(function(cart) {
                var isDonationProduct = false;
                
                // Vérifier si le panier contient le produit Don (ID variante: 52543227003226)
                if (cart.items && cart.items.length > 0) {
                    isDonationProduct = cart.items.some(function(item) {
                        // Vérifier par l'ID de variante
                        if (item.variant_id === 52543227003226) {
                            return true;
                        }
                        
                        // Vérifier par les propriétés
                        if (item.properties) {
                            return Object.keys(item.properties).some(function(key) {
                                return key === 'Montant du don' || 
                                       (item.properties[key] && 
                                        item.properties[key].toString().includes('€'));
                            });
                        }
                        
                        // Vérifier par le titre du produit
                        if (item.product_title && 
                            (item.product_title.toLowerCase().includes('don') ||
                             item.product_title.toLowerCase().includes('cadeau'))) {
                            return true;
                        }
                        
                        return false;
                    });
                    
                    // Vérifier aussi qu'il n'y a QUE le produit Don (pas d'autres produits)
                    var onlyDonationProduct = isDonationProduct && cart.items.length === 1;
                    
                    // Si c'est un don, masquer la section
                    if (onlyDonationProduct) {
                        // Méthode 1 : Masquer le div avec la classe spécifique qui contient shippingMethod
                        var shippingDiv = document.querySelector('div._1ip0g651._1ip0g650._1fragemn2._1fragem50._1fragem6t._1fragem41:has(h3#shippingMethod)');
                        
                        if (!shippingDiv) {
                            // Méthode 2 : Chercher le div qui contient #shippingMethod
                            var heading = document.getElementById('shippingMethod');
                            if (heading) {
                                var div = heading.closest('div._1ip0g651');
                                if (div && div.classList.contains('_1ip0g650') && 
                                    div.classList.contains('_1fragemn2') && 
                                    div.classList.contains('_1fragem50') &&
                                    div.classList.contains('_1fragem6t') &&
                                    div.classList.contains('_1fragem41')) {
                                    shippingDiv = div;
                                } else {
                                    // Remonter jusqu'à trouver le div avec ces classes
                                    var element = heading;
                                    for (var i = 0; i < 10; i++) {
                                        element = element.parentElement;
                                        if (!element || element === document.body) break;
                                        
                                        if (element.tagName === 'DIV' && 
                                            element.classList.contains('_1ip0g651') &&
                                            element.classList.contains('_1ip0g650') &&
                                            element.classList.contains('_1fragem50')) {
                                            shippingDiv = element;
                                            break;
                                        }
                                    }
                                }
                            }
                        }
                        
                        // Masquer le div trouvé
                        if (shippingDiv) {
                            shippingDiv.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important; margin: 0 !important; padding: 0 !important;';
                        }
                        
                        // Méthode 3 : Masquer aussi par le titre et le fieldset
                        var heading = document.getElementById('shippingMethod');
                        if (heading) {
                            var section = heading.closest('section, div._1ip0g651');
                            if (section && section.querySelector('input[name="shipping_methods"]')) {
                                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                            }
                        }
                        
                        var fieldset = document.querySelector('fieldset#shipping_methods');

    // Fonction qui essaie jusqu'à réussir
    function keepTrying() {
        var attempts = 0;
        var maxAttempts = 30;
        
        function tryOnce() {
            attempts++;
            hideShippingForDonation();
            
            if (attempts < maxAttempts) {
                setTimeout(tryOnce, 1000);
            }
        }
        
        tryOnce();
    }
    
    // Exécuter immédiatement
    hideShippingForDonation();

    setTimeout(hideShippingForDonation, 1000);
    setTimeout(hideShippingForDonation, 2000);
    setTimeout(hideShippingForDonation, 5000);
    setTimeout(keepTrying, 3000);
    
    // Attendre que le DOM soit chargé
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', function() {
            hideShippingForDonation();
            setTimeout(keepTrying, 500);
        });
    } else {
        keepTrying();
    }
    
    // Attendre que la page soit complètement chargée
    window.addEventListener('load', function() {
        setTimeout(hideShippingForDonation, 500);
        setTimeout(keepTrying, 1000);
    });
    
    // Observer les changements DOM

                    subtree: true
                });

            }
        }
        setupObserver();
    }
