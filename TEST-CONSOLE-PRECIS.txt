TEST DIRECT DANS LA CONSOLE - Version précise
========================================

Pour tester directement dans la console (F12 > Console), copiez ce code :

========================================

// Test masquage précis - Copiez-collez dans la console
(function() {
    console.log('=== TEST MASQUAGE SECTION EXPEDITION ===');
    
    // Trouver le titre "Mode d'expédition"
    var heading = document.getElementById('shippingMethod');
    console.log('Titre trouvé:', heading);
    
    if (heading) {
        // Remonter jusqu'à la section complète
        var section = heading.parentElement;
        console.log('Section parente directe:', section.tagName, section.className);
        
        // Chercher la section qui contient les options
        for (var i = 0; i < 10; i++) {
            if (!section || section === document.body) break;
            
            if (section.querySelector && section.querySelector('input[name="shipping_methods"]')) {
                console.log('SECTION TROUVÉE (contient inputs shipping):', section.tagName, section.id || section.className);
                console.log('Texte de la section:', section.textContent.substring(0, 200));
                
                // Masquer cette section
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                console.log('✅ SECTION MASQUÉE AVEC SUCCÈS');
                break;
            }
            
            section = section.parentElement;
        }
    }
    
    // Vérifier si le fieldset existe
    var fieldset = document.querySelector('fieldset#shipping_methods');
    if (fieldset) {
        console.log('Fieldset trouvé:', fieldset);
        var section = fieldset.closest('section');
        if (section) {
            console.log('Section parente du fieldset:', section.tagName);
            section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            console.log('✅ SECTION MASQUÉE PAR FIELDSET');
        }
    }
    
    console.log('=== FIN TEST ===');
})();

========================================

INSTRUCTIONS :
1. Allez sur checkout avec un don
2. Ouvrez la console (F12)
3. Collez ce code et appuyez sur Entrée
4. Regardez les messages dans la console
5. Vérifiez si la section disparaît

Si ça marche dans la console mais pas avec le script, le problème vient du timing ou de l'emplacement du script.
