<script>
// Masquer UNIQUEMENT "Mode d'expédition" et les options de livraison
(function() {
    function hideShippingOnly() {
        if (!document.location.href.includes('/checkout')) return;
        
        // Trouver le titre "Mode d'expédition"
        var heading = document.getElementById('shippingMethod');
        if (heading) {
            var section = heading.closest('section, fieldset');
            
            // Si la section contient des inputs shipping et PAS de champs de contact
            if (section && 
                section.querySelector('input[name="shipping_methods"]') &&
                !section.querySelector('input[type="tel"]') &&
                !section.querySelector('input[type="email"]')) {
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            }
        }
        
        // Masquer le fieldset des shipping methods
        var fieldset = document.querySelector('fieldset#shipping_methods');
        if (fieldset) {
            fieldset.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
        }
        
        // Masquer uniquement les options de livraison (pas les autres sections)
        var sections = document.querySelectorAll('section, fieldset');
        sections.forEach(function(sec) {
            var text = sec.textContent || '';
            var hasShipping = (text.includes('Livraison locale') ||
                              text.includes('Livraison en point relais') ||
                              text.includes('Livraison à domicile')) &&
                             sec.querySelector('input[name="shipping_methods"]');
            var hasContact = sec.querySelector('input[type="tel"]') ||
                            sec.querySelector('input[type="email"]') ||
                            sec.querySelector('select[name="countryCode"]');
            
            if (hasShipping && !hasContact) {
                sec.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            }
        });
    }
    
    hideShippingOnly();
    setTimeout(hideShippingOnly, 500);
    setTimeout(hideShippingOnly, 1000);
    setTimeout(hideShippingOnly, 2000);
    
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', hideShippingOnly);
    }
    
    if (typeof MutationObserver !== 'undefined') {
        var obs = new MutationObserver(hideShippingOnly);
        if (document.body) obs.observe(document.body, { childList: true, subtree: true });
    }
})();
</script>
