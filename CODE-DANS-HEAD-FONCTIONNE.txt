<script>
// Version pour le <head> - Attend que le contenu soit chargé
(function() {
    'use strict';
    
    function hideShippingSection() {
        // Vérifier qu'on est sur checkout
        if (!document.location.href.includes('/checkout')) return;
        
        // Vérifier que les éléments existent
        var heading = document.getElementById('shippingMethod');
        var shippingInputs = document.querySelectorAll('input[name="shipping_methods"]');
        
        // Si aucun élément n'existe encore, ne rien faire
        if (!heading && shippingInputs.length === 0) {
            return;
        }
        
        // Méthode 1 : Masquer via le titre "Mode d'expédition"
        if (heading) {
            var section = heading.parentElement;
            
            // Remonter jusqu'à trouver la section qui contient les inputs shipping
            for (var i = 0; i < 10; i++) {
                if (!section || section === document.body) break;
                
                if (section.querySelector && section.querySelector('input[name="shipping_methods"]')) {
                    section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important; margin: 0 !important; padding: 0 !important;';
                    return; // Succès, on arrête
                }
                
                section = section.parentElement;
            }
            
            // Si pas trouvé, essayer avec closest
            section = heading.closest('section');
            if (section && section.querySelector('input[name="shipping_methods"]')) {
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                return;
            }
        }
        
        // Méthode 2 : Masquer via le fieldset
        var shippingFieldset = document.querySelector('fieldset#shipping_methods');
        if (shippingFieldset) {
            var parentSection = shippingFieldset.closest('section');
            if (parentSection) {
                parentSection.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                return;
            }
            shippingFieldset.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            return;
        }
        
        // Méthode 3 : Masquer via les inputs shipping
        if (shippingInputs.length > 0) {
            shippingInputs.forEach(function(input) {
                var section = input.closest('section');
                if (section) {
                    section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                    return;
                }
            });
        }
        
        // Méthode 4 : Masquer via le conteneur .yyi4nyn
        var optionsContainer = document.querySelector('.yyi4nyn.yyi4nyk.yyi4nyr.yyi4nys');
        if (optionsContainer) {
            var section = optionsContainer.closest('section');
            if (section) {
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            }
        }
    }
    
    // Fonction qui essaie plusieurs fois jusqu'à réussir
    function tryHideShipping() {
        var attempts = 0;
        var maxAttempts = 20;
        
        function attempt() {
            attempts++;
            
            // Vérifier si les éléments existent
            var heading = document.getElementById('shippingMethod');
            var shippingInputs = document.querySelectorAll('input[name="shipping_methods"]');
            
            if (heading || shippingInputs.length > 0) {
                hideShippingSection();
                
                // Vérifier si ça a marché
                var headingStillVisible = heading && heading.offsetParent !== null;
                var inputsStillVisible = false;
                shippingInputs.forEach(function(input) {
                    if (input.offsetParent !== null) {
                        inputsStillVisible = true;
                    }
                });
                
                // Si c'est masqué, on arrête
                if (!headingStillVisible && !inputsStillVisible) {
                    return; // Succès !
                }
            }
            
            // Si pas encore masqué et qu'on a encore des tentatives, réessayer
            if (attempts < maxAttempts) {
                setTimeout(attempt, 500);
            }
        }
        
        // Commencer les tentatives
        attempt();
    }
    
    // Attendre que le DOM soit chargé
    function init() {
        // Attendre un peu que le contenu soit chargé
        setTimeout(tryHideShipping, 100);
        setTimeout(tryHideShipping, 500);
        setTimeout(tryHideShipping, 1000);
        setTimeout(tryHideShipping, 2000);
        setTimeout(tryHideShipping, 3000);
        setTimeout(tryHideShipping, 5000);
    }
    
    // Différents événements selon l'état du DOM
    if (document.readyState === 'loading') {
        // DOM pas encore chargé
        document.addEventListener('DOMContentLoaded', init);
    } else if (document.readyState === 'interactive') {
        // DOM chargé mais pas toutes les ressources
        init();
    } else {
        // Tout est chargé
        init();
    }
    
    // Attendre aussi que la page soit complètement chargée
    window.addEventListener('load', function() {
        setTimeout(tryHideShipping, 500);
        setTimeout(tryHideShipping, 1000);
        setTimeout(tryHideShipping, 2000);
    });
    
    // Observer les changements DOM (si Shopify charge le contenu dynamiquement)
    if (typeof MutationObserver !== 'undefined') {
        var observer = new MutationObserver(function(mutations) {
            // Vérifier si de nouveaux éléments de shipping apparaissent
            var heading = document.getElementById('shippingMethod');
            var shippingInputs = document.querySelectorAll('input[name="shipping_methods"]');
            
            if (heading || shippingInputs.length > 0) {
                hideShippingSection();
            }
        });
        
        // Commencer à observer dès que body existe
        if (document.body) {
            observer.observe(document.body, {
                childList: true,
                subtree: true
            });
        } else {
            document.addEventListener('DOMContentLoaded', function() {
                if (document.body) {
                    observer.observe(document.body, {
                        childList: true,
                        subtree: true
                    });
                }
            });
        }
    }
    
    // Exécuter aussi après un long délai (au cas où Shopify charge très tard)
    setTimeout(tryHideShipping, 7000);
    setTimeout(tryHideShipping, 10000);
})();
