<script>
// Masquer UNIQUEMENT la section "Mode d'expédition" et ses options de livraison
(function() {
    function hideShippingSection() {
        if (!document.location.href.includes('/checkout')) return;
        
        // Méthode 1 : Masquer directement le titre "Mode d'expédition" et sa section parente
        var shippingHeading = document.getElementById('shippingMethod');
        if (shippingHeading) {
            // Remonter jusqu'à trouver la section qui contient TOUT (titre + options)
            var section = shippingHeading.parentElement;
            
            // Chercher la section complète qui contient le titre ET les options
            for (var i = 0; i < 8; i++) {
                if (!section || section === document.body) break;
                
                // Si cette section contient des inputs shipping, c'est la bonne section
                if (section.querySelector && section.querySelector('input[name="shipping_methods"]')) {
                    // Masquer TOUTE cette section (y compris les champs de téléphone dedans)
                    section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important; margin: 0 !important; padding: 0 !important;';
                    break;
                }
                
                section = section.parentElement;
            }
            
            // Si on n'a pas trouvé, essayer avec closest
            if (!section || section === document.body || section.style.display !== 'none') {
                section = shippingHeading.closest('section, fieldset, div[data-section-id]');
                if (section && section.querySelector('input[name="shipping_methods"]')) {
                    section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                }
            }
        }
        
        // Méthode 2 : Masquer le fieldset qui contient les options
        var shippingFieldset = document.querySelector('fieldset#shipping_methods');
        if (shippingFieldset) {
            // Remonter jusqu'à la section parente
            var parentSection = shippingFieldset.closest('section');
            if (parentSection) {
                // Vérifier si cette section contient le titre "Mode d'expédition"
                if (parentSection.querySelector('#shippingMethod') || 
                    parentSection.textContent.includes('Mode d') && 
                    parentSection.textContent.includes('expédition')) {
                    parentSection.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                } else {
                    // Masquer juste le fieldset
                    shippingFieldset.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                }
            } else {
                shippingFieldset.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            }
        }
        
        // Méthode 3 : Trouver toutes les sections qui contiennent "Mode d'expédition" dans le titre
        var allSections = document.querySelectorAll('section');
        allSections.forEach(function(sec) {
            var heading = sec.querySelector('h3#shippingMethod, h3:contains("Mode d"), h3:contains("expédition")');
            
            // Si cette section a le titre "Mode d'expédition" ET contient des options de livraison
            if (heading || (sec.querySelector('#shippingMethod') && sec.querySelector('input[name="shipping_methods"]'))) {
                // Masquer TOUTE cette section (y compris tout ce qu'elle contient)
                sec.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            }
        });
        
        // Méthode 4 : Masquer les sections qui contiennent les options de livraison
        // MAIS qui sont après la section "Livraison" (pas avant)
        var shippingInputs = document.querySelectorAll('input[name="shipping_methods"]');
        if (shippingInputs.length > 0) {
            shippingInputs.forEach(function(input) {
                var section = input.closest('section');
                
                if (section) {
                    // Vérifier si cette section contient le titre "Mode d'expédition"
                    var hasShippingTitle = section.querySelector('#shippingMethod') ||
                                         (section.textContent.includes('Mode d') && 
                                          section.textContent.includes('expédition'));
                    
                    // Si c'est une section d'expédition, masquer TOUT
                    if (hasShippingTitle || 
                        section.textContent.includes('Livraison locale') ||
                        section.textContent.includes('Livraison en point relais') ||
                        section.textContent.includes('Livraison à domicile')) {
                        section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                    }
                }
            });
        }
        
        // Méthode 5 : Masquer le conteneur .yyi4nyn et sa section parente
        var optionsContainer = document.querySelector('.yyi4nyn.yyi4nyk.yyi4nyr.yyi4nys');

            var section = optionsContainer.closest('section');
            if (section && (section.querySelector('#shippingMethod') || 
                           section.textContent.includes('Mode d'))) {
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            }
        }
    }
    
    // Exécuter plusieurs fois
    hideShippingSection();
    setTimeout(hideShippingSection, 300);
    setTimeout(hideShippingSection, 600);
    setTimeout(hideShippingSection, 1000);
    setTimeout(hideShippingSection, 2000);
    setTimeout(hideShippingSection, 3000);
    
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', hideShippingSection);
    }
    
    if (typeof MutationObserver !== 'undefined') {
        var observer = new MutationObserver(hideShippingSection);
        if (document.body) {
            observer.observe(document.body, {
                childList: true,
                subtree: true
            });
        }
    }
