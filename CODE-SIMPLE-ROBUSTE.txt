CODE SIMPLE ET ROBUSTE - Pour le <head>
========================================

Cette version essaie toutes les secondes jusqu'à ce que les éléments apparaissent et soient masqués.

========================================

<script>
// Version simple et robuste pour le <head>
(function() {
    'use strict';
    
    function hideShipping() {
        // Vérifier qu'on est sur checkout
        if (!window.location.href.includes('/checkout')) return;
        
        // Vérifier si les éléments existent
        var heading = document.getElementById('shippingMethod');
        var fieldset = document.querySelector('fieldset#shipping_methods');
        var inputs = document.querySelectorAll('input[name="shipping_methods"]');
        
        // Si rien n'existe, on retourne (on réessayera plus tard)
        if (!heading && !fieldset && inputs.length === 0) {
            return false; // Pas encore trouvé
        }
        
        // Masquer via le titre
        if (heading) {
            var section = heading.closest('section');
            if (section && section.querySelector('input[name="shipping_methods"]')) {
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                return true; // Succès
            }
        }
        
        // Masquer via le fieldset
        if (fieldset) {
            var section = fieldset.closest('section');
            if (section) {
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                return true;
            }
            fieldset.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
            return true;
        }
        
        // Masquer via les inputs
        if (inputs.length > 0) {
            inputs.forEach(function(input) {
                var section = input.closest('section');
                if (section) {
                    section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                    return true;
                }
            });
            return true;
        }
        
        return false; // Pas réussi cette fois
    }
    
    // Fonction qui essaie toutes les secondes jusqu'à réussir
    function keepTrying() {
        var success = hideShipping();
        
        // Si pas encore réussi, réessayer dans 1 seconde
        // Mais ne pas dépasser 30 secondes (30 tentatives)
        var attempts = 0;
        var maxAttempts = 30;
        
        var interval = setInterval(function() {
            attempts++;
            var success = hideShipping();
            
            if (success || attempts >= maxAttempts) {
                clearInterval(interval);
            }
        }, 1000);
    }
    
    // Exécuter immédiatement
    hideShipping();
    
    // Exécuter après différents délais
    setTimeout(keepTrying, 500);
    setTimeout(keepTrying, 1000);
    setTimeout(keepTrying, 2000);
    setTimeout(keepTrying, 5000);
    
    // Attendre que le DOM soit chargé
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', keepTrying);
    } else {
        keepTrying();
    }
    
    // Attendre que la page soit complètement chargée
    window.addEventListener('load', keepTrying);
    
    // Observer les changements DOM
    if (typeof MutationObserver !== 'undefined' && document.body) {
        var observer = new MutationObserver(function() {
            hideShipping();
        });
        observer.observe(document.body, {
            childList: true,
            subtree: true
        });
    }
})();
</script>
