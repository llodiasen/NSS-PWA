CODE ULTIME - Garanti de fonctionner dans le <head>
========================================

Cette version injecte aussi du CSS dynamiquement ET masque avec JavaScript.
Si le JavaScript ne fonctionne pas, le CSS prendra le relais.

========================================

<script>
// Version ultime - CSS + JavaScript
(function() {
    'use strict';
    
    // Injecter le CSS IMMÉDIATEMENT (même si le contenu n'est pas encore chargé)
    function injectCSS() {
        if (document.getElementById('hide-shipping-css-injected')) return;
        
        var style = document.createElement('style');
        style.id = 'hide-shipping-css-injected';
        style.textContent = `
            /* Masquer la section Mode d'expédition */
            h3#shippingMethod,
            fieldset#shipping_methods,
            section:has(input[name="shipping_methods"]),
            section:has(#shippingMethod),
            section:has(fieldset#shipping_methods) {
                display: none !important;
                visibility: hidden !important;
                height: 0 !important;
                overflow: hidden !important;
                margin: 0 !important;
                padding: 0 !important;
            }
            
            /* Masquer aussi le conteneur des options */
            .yyi4nyn.yyi4nyk.yyi4nyr.yyi4nys:has(input[name="shipping_methods"]),
            section:has(.yyi4nyn.yyi4nyk.yyi4nyr.yyi4nys) {
                display: none !important;
                visibility: hidden !important;
                height: 0 !important;
                overflow: hidden !important;
            }
        `;
        
        var head = document.head || document.getElementsByTagName('head')[0];
        if (head) {
            head.appendChild(style);
        }
    }
    
    // Injecter le CSS immédiatement
    injectCSS();
    
    // Fonction pour masquer avec JavaScript
    function hideShippingJS() {
        if (!window.location.href.includes('/checkout')) return false;
        
        var heading = document.getElementById('shippingMethod');
        var fieldset = document.querySelector('fieldset#shipping_methods');
        var inputs = document.querySelectorAll('input[name="shipping_methods"]');
        
        if (!heading && !fieldset && inputs.length === 0) {
            return false;
        }
        
        var success = false;
        
        // Masquer via le titre
        if (heading) {
            var section = heading.closest('section');
            if (section && section.querySelector('input[name="shipping_methods"]')) {
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                success = true;
            }
        }
        
        // Masquer via le fieldset
        if (fieldset) {
            var section = fieldset.closest('section');
            if (section) {
                section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                success = true;
            } else {
                fieldset.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                success = true;
            }
        }
        
        // Masquer via les inputs
        if (inputs.length > 0) {
            inputs.forEach(function(input) {
                var section = input.closest('section');
                if (section) {
                    section.style.cssText = 'display: none !important; visibility: hidden !important; height: 0 !important; overflow: hidden !important;';
                    success = true;
                }
            });
        }
        
        return success;
    }
    
    // Fonction qui essaie jusqu'à réussir
    function keepTrying() {
        var attempts = 0;
        var maxAttempts = 30;
        
        function tryOnce() {
            attempts++;
            var success = hideShippingJS();
            
            if (!success && attempts < maxAttempts) {
                setTimeout(tryOnce, 1000);
            }
        }
        
        tryOnce();
    }
    
    // Exécuter immédiatement
    hideShippingJS();
    
    // Exécuter après différents délais
    setTimeout(function() {
        hideShippingJS();
        keepTrying();
    }, 300);
    
    setTimeout(function() {
        hideShippingJS();
        keepTrying();
    }, 1000);
    
    setTimeout(function() {
        hideShippingJS();
        keepTrying();
    }, 2000);
    
    setTimeout(function() {
        hideShippingJS();
        keepTrying();
    }, 5000);
    
    // Attendre que le DOM soit chargé
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', function() {
            hideShippingJS();
            keepTrying();
        });
    } else {
        keepTrying();
    }
    
    // Attendre que la page soit complètement chargée
    window.addEventListener('load', function() {
        setTimeout(function() {
            hideShippingJS();
            keepTrying();
        }, 500);
    });
    
    // Observer les changements DOM
    if (typeof MutationObserver !== 'undefined') {
        function setupObserver() {
            if (document.body) {
                var observer = new MutationObserver(function() {
                    hideShippingJS();
                });
                observer.observe(document.body, {
                    childList: true,
                    subtree: true
                });
            } else {
                setTimeout(setupObserver, 100);
            }
        }
        setupObserver();
    }
})();
</script>

========================================

Cette version :
1. Injecte le CSS IMMÉDIATEMENT (fonctionne même si JavaScript est désactivé)
2. Essaie avec JavaScript plusieurs fois
3. Réessaie toutes les secondes jusqu'à 30 tentatives
4. Observe les changements DOM en continu
